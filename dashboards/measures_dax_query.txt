// UC15 ERP dashboard: all measures in one script for Power BI Desktop's DAX query view.
// How to use: View > DAX query view > paste this whole text > click 'Update model with changes' (it appears above the editor).
// If your version has no DAX query view, create each measure by hand with Home > New measure. The same measures are listed in dashboards/Dashboard_Design.md, Section 5.
// The table names must match the CSV file names (vw_dash_sales and so on), which is what Power BI uses when you import them.

DEFINE
    MEASURE vw_dash_sales[Total Revenue] = SUM ( vw_dash_sales[total_amount] )
    MEASURE vw_dash_sales[Sales Transactions] = COUNTROWS ( vw_dash_sales )
    MEASURE vw_dash_sales[Avg Order Value] = DIVIDE ( [Total Revenue], [Sales Transactions] )
    MEASURE vw_dash_sales[Units Sold] = SUM ( vw_dash_sales[quantity] )
    MEASURE vw_dash_procurement[Spend (excl. cancelled)] = SUM ( vw_dash_procurement[spend_excl_cancelled] )
    MEASURE vw_dash_procurement[Ordered Value (all orders)] = SUM ( vw_dash_procurement[total_cost] )
    MEASURE vw_dash_procurement[Purchase Orders] = COUNTROWS ( vw_dash_procurement )
    MEASURE vw_dash_procurement[Cancelled Cost Share] = DIVIDE ( CALCULATE ( SUM ( vw_dash_procurement[total_cost] ), vw_dash_procurement[is_cancelled] = 1 ), [Ordered Value (all orders)] )
    MEASURE vw_dash_receivables[Receivable Amount] = SUM ( vw_dash_receivables[amount] )
    MEASURE vw_dash_receivables[Receivable Documents] = COUNTROWS ( vw_dash_receivables )
    MEASURE vw_dash_receivables[Delay Rate] = DIVIDE ( CALCULATE ( COUNTROWS ( vw_dash_receivables ), vw_dash_receivables[delayflag] = 1 ), [Receivable Documents] )
    MEASURE vw_dash_receivables[Consistent Dates Share] = DIVIDE ( CALCULATE ( COUNTROWS ( vw_dash_receivables ), vw_dash_receivables[dates_consistent] = 1 ), [Receivable Documents] )
    MEASURE vw_dash_inventory_demand[Units Demanded] = SUM ( vw_dash_inventory_demand[units_demanded] )
    MEASURE vw_dash_warehouse[Items Below Reorder Point] = CALCULATE ( COUNTROWS ( vw_dash_warehouse ), vw_dash_warehouse[below_reorder_point] = 1 )
    MEASURE vw_dash_warehouse[Avg Fulfilment Rate] = AVERAGE ( vw_dash_warehouse[order_fulfillment_rate] )
    MEASURE vw_dash_warehouse[Avg Turnover Ratio] = AVERAGE ( vw_dash_warehouse[turnover_ratio] )
    MEASURE vw_dash_warehouse[Stock Value] = SUM ( vw_dash_warehouse[stock_value] )
    MEASURE vw_dash_warehouse[Stockouts Last Month] = SUM ( vw_dash_warehouse[stockout_count_last_month] )
    MEASURE vw_dash_manufacturing[Jobs] = COUNTROWS ( vw_dash_manufacturing )
    MEASURE vw_dash_manufacturing[Failed Job Rate] = DIVIDE ( SUM ( vw_dash_manufacturing[is_failed] ), [Jobs] )
    MEASURE vw_dash_manufacturing[Completed Job Rate] = DIVIDE ( SUM ( vw_dash_manufacturing[is_completed] ), [Jobs] )
    MEASURE vw_dash_manufacturing[Avg Energy per Job] = AVERAGE ( vw_dash_manufacturing[energy_consumption] )
    MEASURE vw_dash_manufacturing[Low Efficiency Share] = DIVIDE ( CALCULATE ( [Jobs], vw_dash_manufacturing[optimization_category] = "Low Efficiency" ), [Jobs] )
    MEASURE vw_dash_assets[Obsolete Asset Share] = DIVIDE ( CALCULATE ( COUNTROWS ( vw_dash_assets ), vw_dash_assets[operational_state] = "Obsolete" ), COUNTROWS ( vw_dash_assets ) )
    MEASURE vw_dash_assets[Maintenance Cost] = SUM ( vw_dash_assets[maintenance_cost_last_year] )
    MEASURE vw_dash_assets[Avg Condition Score] = AVERAGE ( vw_dash_assets[condition_score] )
    MEASURE vw_dash_customers[Customers] = COUNTROWS ( vw_dash_customers )
    MEASURE vw_dash_customers[Active Customer Share] = DIVIDE ( CALCULATE ( [Customers], vw_dash_customers[customer_status] = "Active" ), [Customers] )
    MEASURE vw_dash_customers[Credit Limit Total] = SUM ( vw_dash_customers[credit_limit] )
    MEASURE vw_dash_employees[Avg Salary] = AVERAGE ( vw_dash_employees[salary] )
    MEASURE vw_dash_finance[Finance Revenue (as given)] = SUM ( vw_dash_finance[revenue] )
    MEASURE vw_dash_finance[Finance Expenditure (as given)] = SUM ( vw_dash_finance[expenditure] )
    MEASURE vw_dash_finance[Finance Amount (as given)] = SUM ( vw_dash_finance[transaction_amount] )
    MEASURE vw_dash_data_checks[Checks Passed] = CALCULATE ( COUNTROWS ( vw_dash_data_checks ), vw_dash_data_checks[status] = "PASS" )
    MEASURE vw_dash_data_checks[Known Issues] = CALCULATE ( COUNTROWS ( vw_dash_data_checks ), vw_dash_data_checks[status] = "KNOWN ISSUE" )
    MEASURE vw_dash_data_checks[Unexpected] = CALCULATE ( COUNTROWS ( vw_dash_data_checks ), vw_dash_data_checks[status] = "UNEXPECTED" )

EVALUATE ROW ( "measures_in_this_script", 36 )
